Usage Example

import { createDelegator, createHandlerPlugin } from './src/delegator.js';
import { copyTextPlugin, copyLinkPlugin } from './src/plugins/copy.js';

// Create delegator with ignore zones
const delegator = createDelegator({
  root: document,
  ignore: '[data-ignore]',
});

// Add handler plugin
delegator.use(createHandlerPlugin({
  handlers: {
    sayHello: () => alert('Hello!'),
    Demo: {
      increment: () => count++,
      decrement: () => count--,
    }
  }
}));

// Add copy plugins
delegator.use(copyTextPlugin());
delegator.use(copyLinkPlugin());

// Start listening
delegator.start();
